    <div class="section load-section">
        <h3>Загрузка существующего data.json</h3>
        <input type="file" id="loadJsonFile" accept=".json" onchange="loadExistingJson()">
        <button onclick="clearForm()">Очистить форму</button>
    </div>

    <div class="section">
        <h3>Основная информация</h3>
        <label>Название урока: <input type="text" id="lessonTitle" placeholder="Например: Урок 1: Paradise city"></label>
        <label>Описание (HTML): <textarea id="lessonContent" placeholder="<p>Описание урока...</p>"></textarea></label>
    </div>

    <div class="section">
        <h3>Партитуры</h3>
        <button onclick="addScore()">Добавить партитуру</button>
        <div id="scoresContainer"></div>
    </div>

    <div class="section">
        <h3>Сгенерированный JSON</h3>
        <textarea id="jsonOutput" class="json-output" readonly></textarea>
        <button onclick="copyToClipboard()">Скопировать в буфер обмена</button>
        <button onclick="downloadJSON()">Скачать файл</button>

    <script>
        let scoreCount = 0;
        // Изменён порядок дорожек: kick всегда последний
        const availableTracks = ['crash', 'hihat', 'snare', 'tom1', 'tom2', 'tom3', 'kick'];

        function addScore(initialData = {}) {
            const container = document.getElementById('scoresContainer');
            const scoreId = `score-${scoreCount++}`;
            const scoreDiv = document.createElement('div');
            scoreDiv.className = 'score-item';
            scoreDiv.id = scoreId;
            scoreDiv.draggable = true;

            // Инициализация с данными, если они предоставлены
            const title = initialData.title || '';
            const description = initialData.description || '';
            const url = initialData.url || '';
            const hasSequencer = initialData.sequencer !== false;
            const sequencerTracks = initialData.sequencer && Array.isArray(initialData.sequencer.tracks) ? initialData.sequencer.tracks : [];
            const sequencerTargetCells = initialData.sequencer && initialData.sequencer.targetCells ? initialData.sequencer.targetCells : {};

            // Определяем класс для каждой ячейки по битам
            function getBeatClass(cellNum) {
                const beat = Math.floor((cellNum - 1) / 4) + 1;
                return `beat-${beat}`;
            }

            scoreDiv.innerHTML = `
                <div class="score-header">
                    🎵 Партитура #${scoreCount} <span class="toggle-icon">▼</span>
                </div>
                <div class="score-content" style="display: block;">
                    <label>Название: <input type="text" name="title" placeholder="Вступление" value="${escapeHtml(title)}"></label>
                    <label>Описание: <textarea name="description" placeholder="Описание партитуры...">${escapeHtml(description)}</textarea></label>
                    <label>Путь к .musicxml: <input type="text" name="url" placeholder="lessons/lesson6/0.musicxml" value="${escapeHtml(url)}"></label>
                    <label><input type="checkbox" name="hasSequencer" onchange="toggleSequencer(this)" ${hasSequencer ? 'checked' : ''}> Включить секвенсор</label>
                    <div class="sequencer-section" style="display:${hasSequencer ? 'block' : 'none'};" name="sequencerSection">
                        <div class="tracks-selector">
                            <strong>Видимые дорожки:</strong>
                            ${availableTracks.map(track => `
                                <div class="track-option">
                                    <input type="checkbox" name="track-${track}" ${sequencerTracks.includes(track) ? 'checked' : ''}>
                                    <label>${track}</label>
                                </div>
                            `).join('')}
                        </div>
                        <div class="cells-container">
                            <strong>Целевые ячейки:</strong>
                            ${availableTracks.map(track => `
                                <div class="track-cells" data-track="${track}" style="margin: 10px 0; ${sequencerTracks.includes(track) ? '' : 'display: none;'}">
                                    <div class="cell-label">${track}:</div>
                                    <div class="cells-wrapper" data-track="${track}">
                                        ${Array.from({length: 4}, (_, groupIndex) => `
                                            <div class="cells-group">
                                                ${Array.from({length: 4}, (_, i) => {
                                                    const cellNum = groupIndex * 4 + i + 1;
                                                    return `
                                                        <div class="cell ${getBeatClass(cellNum)} ${sequencerTargetCells[track] && sequencerTargetCells[track].includes(cellNum) ? 'active' : ''}" 
                                                             data-cell="${cellNum}" 
                                                             data-beat="${Math.floor((cellNum - 1) / 4) + 1}"
                                                             onclick="toggleCell(this, '${track}')">${cellNum}</div>
                                                    `;
                                                }).join('')}
                                            </div>
                                        `).join('')}
                                    </div>
                                </div>
                            `).join('')}
                        </div>
                    </div>
                    <button type="button" class="remove" onclick="removeScore('${scoreId}')">Удалить</button>
                </div>
            `;

            // Добавляем обработчики drag'n'drop
            scoreDiv.addEventListener('dragstart', handleDragStart);
            scoreDiv.addEventListener('dragover', handleDragOver);
            scoreDiv.addEventListener('dragenter', handleDragEnter);
            scoreDiv.addEventListener('dragleave', handleDragLeave);
            scoreDiv.addEventListener('drop', handleDrop);
            scoreDiv.addEventListener('dragend', handleDragEnd);

            // Добавляем обработчики для чекбоксов дорожек
            setTimeout(() => {
                const trackCheckboxes = scoreDiv.querySelectorAll('input[name^="track-"]');
                trackCheckboxes.forEach(checkbox => {
                    checkbox.addEventListener('change', function() {
                        const trackName = this.name.replace('track-', '');
                        const trackCells = scoreDiv.querySelector(`.track-cells[data-track="${trackName}"]`);
                        if (trackCells) {
                            trackCells.style.display = this.checked ? 'block' : 'none';
                        }
                        updateJSONOutput();
                    });
                });
            }, 0);

            // Добавляем обработчик сворачивания
            const header = scoreDiv.querySelector('.score-header');
            const content = scoreDiv.querySelector('.score-content');
            const icon = header.querySelector('.toggle-icon');

            header.addEventListener('click', (e) => {
                if (e.target !== header) return; // Игнорируем клики по дочерним элементам
                if (content.style.display === 'none') {
                    content.style.display = 'block';
                    icon.textContent = '▼';
                } else {
                    content.style.display = 'none';
                    icon.textContent = '▲';
                }
            });

            container.appendChild(scoreDiv);
        }

        // Функция для переключения состояния ячейки
        function toggleCell(cellElement, trackName) {
            cellElement.classList.toggle('active');
            updateJSONOutput();
        }

        // Функция для экранирования HTML (чтобы безопасно вставлять данные в input/textarea)
        function escapeHtml(unsafe) {
            if (typeof unsafe !== 'string') return '';
            return unsafe
                .replace(/&/g, "&amp;")
                .replace(/</g, "<")
                .replace(/>/g, ">")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }

        function toggleSequencer(checkbox) {
            const section = checkbox.closest('.score-item').querySelector('[name="sequencerSection"]');
            section.style.display = checkbox.checked ? 'block' : 'none';
        }

        // --- Переменные для отслеживания drag'n'drop ---
        let draggedElement = null;

        function handleDragStart(e) {
            draggedElement = this;
            e.dataTransfer.effectAllowed = 'move';
            this.style.opacity = '0.7';
        }

        function handleDragOver(e) {
            e.preventDefault(); // Необходимо для drop
            e.dataTransfer.dropEffect = 'move';
            return false;
        }

        function handleDragEnter(e) {
            this.classList.add('drag-over');
        }

        function handleDragLeave(e) {
            this.classList.remove('drag-over');
        }

        function handleDrop(e) {
            e.stopPropagation(); // Останавливаем всплытие

            if (draggedElement !== this) {
                const parentContainer = document.getElementById('scoresContainer');
                const allItems = Array.from(parentContainer.children);

                const fromIndex = allItems.indexOf(draggedElement);
                const toIndex = allItems.indexOf(this);

                if (fromIndex < toIndex) {
                    parentContainer.insertBefore(draggedElement, this.nextSibling);
                } else {
                    parentContainer.insertBefore(draggedElement, this);
                }

                updateJSONOutput();
            }

            this.classList.remove('drag-over');
            return false;
        }

        function handleDragEnd(e) {
            draggedElement.style.opacity = '1';
            draggedElement = null;
        }
        // --- /Конец drag'n'drop ---


        function removeScore(id) {
            const element = document.getElementById(id);
            if (element) element.remove();
        }

        function generateJSON() {
            const data = {
                title: document.getElementById('lessonTitle').value,
                content: document.getElementById('lessonContent').value,
                scores: []
            };

            const scoreElements = document.querySelectorAll('#scoresContainer .score-item');
            scoreElements.forEach(el => {
                const score = {
                    title: el.querySelector('input[name="title"]').value,
                    description: el.querySelector('textarea[name="description"]').value,
                    url: el.querySelector('input[name="url"]').value,
                };

                const hasSequencer = el.querySelector('input[name="hasSequencer"]').checked;
                if (hasSequencer) {
                    const tracks = [];
                    const trackCheckboxes = el.querySelectorAll('input[name^="track-"]');
                    trackCheckboxes.forEach(checkbox => {
                        if (checkbox.checked) {
                            const trackName = checkbox.name.replace('track-', '');
                            tracks.push(trackName);
                        }
                    });

                    const targetCells = {};
                    tracks.forEach(track => {
                        const activeCells = el.querySelectorAll(`.cells-wrapper[data-track="${track}"] .cell.active`);
                        if (activeCells.length > 0) {
                            targetCells[track] = Array.from(activeCells).map(cell => parseInt(cell.dataset.cell));
                        }
                    });

                    score.sequencer = {
                        tracks: tracks,
                        targetCells: targetCells
                    };
                } else {
                    score.sequencer = false;
                }
                data.scores.push(score);
            });

            return JSON.stringify(data, null, 2);
        }

        function updateJSONOutput() {
            const jsonStr = generateJSON();
            document.getElementById('jsonOutput').value = jsonStr;
        }

        // Функция для настройки обработчиков у новых партитур
        function setupScoreEventListeners() {
            document.querySelectorAll('#scoresContainer input, #scoresContainer textarea').forEach(el => {
                el.removeEventListener('input', updateJSONOutput); // Убираем дубликаты
                el.addEventListener('input', updateJSONOutput);
            });
        }

        // Перехватываем добавление новой партитуры
        const originalAddScore = addScore;
        addScore = function(initialData) {
            originalAddScore(initialData);
            setupScoreEventListeners();
        };

        // Инициализация
        setupScoreEventListeners();
        document.getElementById('lessonTitle').addEventListener('input', updateJSONOutput);
        document.getElementById('lessonContent').addEventListener('input', updateJSONOutput);

        function copyToClipboard() {
            const output = document.getElementById('jsonOutput');
            output.select();
            document.execCommand('copy');
            alert('JSON скопирован в буфер обмена!');
        }

        function downloadJSON() {
            const jsonStr = generateJSON();
            const blob = new Blob([jsonStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'data.json';
            a.click();
            URL.revokeObjectURL(url);
        }

        // --- НОВЫЕ ФУНКЦИИ ДЛЯ ЗАГРУЗКИ И ОЧИСТКИ ---

        function loadExistingJson() {
            const fileInput = document.getElementById('loadJsonFile');
            const file = fileInput.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const jsonData = JSON.parse(e.target.result);

                    // Очищаем текущую форму
                    clearForm();

                    // Заполняем основную информацию
                    document.getElementById('lessonTitle').value = jsonData.title || '';
                    document.getElementById('lessonContent').value = jsonData.content || '';

                    // Заполняем партитуры
                    if (jsonData.scores && Array.isArray(jsonData.scores)) {
                        jsonData.scores.forEach(score => {
                            addScore(score); // Передаём данные партитуры в addScore
                        });
                    }

                    // Обновляем JSON в поле вывода
                    updateJSONOutput();

                } catch (error) {
                    alert(`Ошибка при загрузке или парсинге JSON: ${error.message}`);
                    console.error(error);
                }
            };
            reader.readAsText(file);
        }

        function clearForm() {
            document.getElementById('lessonTitle').value = '';
            document.getElementById('lessonContent').value = '';
            document.getElementById('scoresContainer').innerHTML = '';
            document.getElementById('jsonOutput').value = '';
            scoreCount = 0; // Сбросим счётчик, чтобы новые партитуры начинались с 1
        }

    </script>
